# This causes PowerShell to ask me to log in. It creates a credential object in PowerShell.
# -Message is what I see. The message is for me and it should ask me to enter the PAT.
# MUST enter THE PAT here, NOT GitHub password.
$credential = Get-Credential `
    -UserName "labswal" `
    -Message "Enter your classic GitHub PAT as the password"

# This takes the password stored inside the credential object and puts it in a string named $pat.
$pat = $credential.GetNetworkCredential().Password

# This builds a text string and stores it in $credentialValue.
$credentialValue =
    "Username=$($credential.UserName);" +
    "Password=$pat;" +
    "ValidAuthenticationTypes=Basic"

# This sets up a PowerShell Process-level environment variable.
# This is a variable scoped only to this particular instance of PowerShell.
# We need this step because it seems the LA-WLF source has issues if we try to do this other ways.
[Environment]::SetEnvironmentVariable(
    "NuGetPackageSourceCredentials_LA-WLF",
    $credentialValue,
    "Process"
)

# Reads the Process-level environment variable and stores it in a normal $value variable.
$value = [Environment]::GetEnvironmentVariable(
    "NuGetPackageSourceCredentials_LA-WLF",
    "Process"
)

# This should echo back True. Checks to see if $value is null.
Write-Host "NuGet credential environment variable has value: $($value -ne $null)"

# Checks whether $value has the structure we want.
if ($value -match '^Username=.+;Password=.+;ValidAuthenticationTypes=Basic$') {
    Write-Host "Environment variable structure is correct"
}
else {
    Write-Host "Environment variable structure is NOT correct"
}

# Displays $value to the PowerShell user, but hides the PAT.
# Probably overkill, but whatever.
$value -replace 'Password=[^;]+', 'Password=***'

$response = Invoke-WebRequest `

    -Credential $credential

$response.StatusCode
